#!/usr/bin/env bash
# scan_cidrs_rootless.sh
# Usage: ./scan_cidrs_rootless.sh <cidr_file>
# Reads CIDR entries from file and runs nmap -sV (non-root) for each entry.
set -uo pipefail

CIDR_FILE="${1:-}"
if [[ -z "$CIDR_FILE" ]]; then
  echo "Usage: $0 <cidr_file>"
  exit 2
fi

if [[ ! -f "$CIDR_FILE" ]]; then
  echo "File not found: $CIDR_FILE"
  exit 3
fi

OUTDIR="results"
mkdir -p "$OUTDIR"

# simple CIDR validation regex
cidr_re='^([0-9]{1,3}\.){3}[0-9]{1,3}\/([0-9]|[12][0-9]|3[0-2])$'

echo "Reading CIDRs from: $CIDR_FILE"
echo "Results will be stored in: $OUTDIR/"
echo "Running without root privileges."

line_no=0
while IFS= read -r rawline || [[ -n "$rawline" ]]; do
  ((line_no++))
  # remove comments and whitespace
  line="${rawline%%#*}"
  line="$(echo -n "$line" | tr -d $'\t\r\n ')"

  if [[ -z "$line" ]]; then
    continue
  fi

  if [[ ! "$line" =~ $cidr_re ]]; then
    echo "Skipping invalid CIDR on line $line_no: '$rawline'"
    continue
  fi

  safe_name="${line//\//_}"
  outfile="$OUTDIR/${safe_name}.nmap"
  echo "[$line_no] Scanning $line  -> $outfile"

  # run nmap -sV without sudo
  if ! nmap -sV "$line" -oN "$outfile"; then
    echo "  ERROR: nmap failed for $line"
  else
    echo "  Done: $outfile"
  fi

done < "$CIDR_FILE"

echo "All done."
